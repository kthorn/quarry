# Unify User-Posting State — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace `user_labels` and `user_posting_status` tables with a single `user_posting_state` table (nullable boolean `interest`, boolean `applied`).

**Architecture:** Migration creates new table, drops old ones. ORM model, Pydantic model, and DB layer replace string-based signals with boolean interest. UI removes status tabs, simplifies to interest + applied badges. Classifier training switches from `"positive"`/`"negative"` strings to `True`/`False` booleans.

**Tech Stack:** SQLAlchemy ORM, Alembic migrations, SQLite, Flask, Jinja2 templates, scikit-learn LogisticRegression.

    def set_interest(
        self, posting_id: int, value: bool | None, user_id: int = 1
    ) -> None:
        """Set interest for a posting. value=None means clear (unevaluated).

        Also increments labels_since_last_train and checks retrain threshold.
        """
        from quarry.store.models import UserPostingState as ORMState
        from quarry.store.models import UserSetting as ORMUserSetting

        with session_scope(engine=self.engine) as session:
            # Upsert the interest value
            session.execute(
                sqlite_insert(ORMState)
                .values(
                    user_id=user_id,
                    posting_id=posting_id,
                    interest=value,
                    labeled_at=func.now(),
                    updated_at=func.now(),
                )
                .on_conflict_do_update(
                    index_elements=["user_id", "posting_id"],
                    set_=dict(interest=value, labeled_at=func.now(), updated_at=func.now()),
                )
            )

            if value is None:
                return  # clearing interest doesn't count as a training label action

            # Increment labels_since_last_train
            row = session.execute(
                select(ORMUserSetting.value).where(
                    ORMUserSetting.user_id == user_id,
                    ORMUserSetting.key == "labels_since_last_train",
                )
            ).scalar_one_or_none()
            current = int(row) if row and row.isdigit() else 0
            new_count = current + 1

            # Read threshold
            threshold_row = session.execute(
                select(ORMUserSetting.value).where(
                    ORMUserSetting.user_id == user_id,
                    ORMUserSetting.key == "retrain_label_threshold",
                )
            ).scalar_one_or_none()
            threshold = (
                int(threshold_row)
                if threshold_row and threshold_row.isdigit()
                else 20
            )

            # Upsert counter
            session.execute(
                sqlite_insert(ORMUserSetting)
                .values(
                    user_id=user_id,
                    key="labels_since_last_train",
                    value=str(new_count),
                    updated_at=func.now(),
                )
                .on_conflict_do_update(
                    index_elements=["user_id", "key"],
                    set_=dict(value=str(new_count), updated_at=func.now()),
                )
            )

            # Set retrain_pending if threshold reached
            if new_count >= threshold:
                session.execute(
                    sqlite_insert(ORMUserSetting)
                    .values(
                        user_id=user_id,
                        key="retrain_pending",
                        value="true",
                        updated_at=func.now(),
                    )
                    .on_conflict_do_update(
                        index_elements=["user_id", "key"],
                        set_=dict(value="true", updated_at=func.now()),
                    )
                )
```

def test_user_posting_state_crud(seeded):
    """UserPostingState: create, read, update interest/applied."""
    from quarry.store.models import JobPosting, UserPostingState

    session = seeded.db.Session()
    # Get first posting
    posting = session.query(JobPosting).first()
    posting_id = posting.id

    # Initially no state row exists
    state = session.get(UserPostingState, (1, posting_id))
    assert state is None

    # Set interest=True
    state = UserPostingState(
        user_id=1, posting_id=posting_id, interest=True, applied=False
    )
    session.add(state)
    session.commit()

    # Read back
    state = session.get(UserPostingState, (1, posting_id))
    assert state.interest is True
    assert state.applied is False

    # Setting interest=False replaces interest
    state.interest = False
    session.commit()
    state = session.get(UserPostingState, (1, posting_id))
    assert state.interest is False

    # Can also set applied
    state.applied = True
    session.commit()
    state = session.get(UserPostingState, (1, posting_id))
    assert state.applied is True
    assert state.interest is False

    # Setting interest=None is allowed
    state.interest = None
    session.commit()
    state = session.get(UserPostingState, (1, posting_id))
    assert state.interest is None
```
